from __future__ import annotations

from datetime import datetime

from pydantic import BaseModel, ConfigDict, Field, field_validator


class ManagementClientInput(BaseModel):
    client_code: str = Field(min_length=2, max_length=32)
    client_name: str = Field(min_length=2, max_length=255)
    client_email: str | None = Field(default=None, max_length=255)
    organization_email: str | None = Field(default=None, max_length=255)
    address: str | None = Field(default=None, max_length=3000)
    country: str = Field(default="India", min_length=2, max_length=100)
    gst_number: str | None = Field(default=None, max_length=32)
    contact_person_name: str = Field(min_length=2, max_length=255)
    contact_person_email: str | None = Field(default=None, max_length=255)
    contact_person_phone: str | None = Field(default=None, max_length=40)
    work_description: str | None = Field(default=None, max_length=5000)

    @field_validator("client_code", mode="before")
    @classmethod
    def normalize_client_code(cls, value):
        if not isinstance(value, str):
            return value
        value = value.strip().upper()
        allowed = set("ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789-_/. ")
        if any(character not in allowed for character in value):
            raise ValueError("Client ID contains unsupported characters")
        return value

    @field_validator(
        "client_name", "client_email", "organization_email", "address", "country",
        "gst_number", "contact_person_name", "contact_person_email", "contact_person_phone",
        "work_description", mode="before",
    )
    @classmethod
    def strip_text(cls, value):
        if isinstance(value, str):
            return value.strip() or None
        return value

    @field_validator("client_email", "organization_email", "contact_person_email")
    @classmethod
    def validate_email(cls, value):
        if not value:
            return value
        value = value.lower()
        local, separator, domain = value.partition("@")
        if not separator or not local or not domain or "." not in domain or domain.startswith(".") or domain.endswith("."):
            raise ValueError("Enter a valid email address")
        return value

    @field_validator("gst_number")
    @classmethod
    def normalize_gst(cls, value):
        return value.upper().replace(" ", "") if value else value


class ManagementClientResponse(ManagementClientInput):
    model_config = ConfigDict(from_attributes=True)
    can_request_project: bool = False

    id: int
    source_person_name: str
    is_active: bool
    created_at: datetime
    updated_at: datetime

# The two authoritative management logins. This is enforced server-side, not only in the sidebar.
MANAGEMENT_EMAILS = frozenset({"vinod@nakshatech.com", "chethan@nakshatech.com"})


def _management_only(auth: CurrentAuth) -> None:
    role = str(auth.effective_role or "").strip().lower()
    email = str(auth.user.email or "").strip().lower()
    if role != "management" or email not in MANAGEMENT_EMAILS:
        raise HTTPException(status_code=status.HTTP_403_FORBIDDEN, detail="Management client register access denied")


def _check_code(db: Session, client_code: str, exclude_id: int | None = None) -> None:
    existing = db.scalar(select(ManagementClientIntake.id).where(ManagementClientIntake.client_code == client_code))
    if existing is not None and existing != exclude_id:
        raise HTTPException(status_code=409, detail="Client ID is already registered in Management")
    # Avoid an identifier collision when Finance later converts this intake into the Client Master.
    finance_existing = db.scalar(select(FinanceClient).where(FinanceClient.client_code == client_code))
    # Finance creates a linked management-sourced FinanceClient on the first project.
    # That linked copy must not block normal edits to the ORIGINAL management intake.
    same_intake = db.get(ManagementClientIntake, exclude_id) if exclude_id is not None else None
    if finance_existing is not None and not (
        same_intake is not None and same_intake.client_code == client_code
        and finance_existing.source_team == "management"
    ):
        raise HTTPException(status_code=409, detail="Client ID already exists in the Finance Client Master")


def _values(payload: ManagementClientInput) -> dict:
    return payload.model_dump()


@router.get("", response_model=list[ManagementClientResponse])
def list_management_clients(
    db: Session = Depends(get_db), auth: CurrentAuth = Depends(get_current_auth),
):
    _management_only(auth)
    rows = db.scalars(
        select(ManagementClientIntake).order_by(
            ManagementClientIntake.created_at.desc(), ManagementClientIntake.id.desc()
        )
    ).all()
    return [ManagementClientResponse.model_validate(row).model_copy(
        update={"can_request_project": row.created_by_id == auth.user.id and row.is_active},
    ) for row in rows]


@router.post("", response_model=ManagementClientResponse, status_code=status.HTTP_201_CREATED)
def create_management_client(
    payload: ManagementClientInput, request: Request,
    db: Session = Depends(get_db), auth: CurrentAuth = Depends(get_current_auth),
):
    _management_only(auth)
    _check_code(db, payload.client_code)
    row = ManagementClientIntake(
        **_values(payload),
        source_person_name=auth.user.full_name,
        created_by_id=auth.user.id,
        updated_by_id=auth.user.id,
        is_active=True,
    )
    try:
        db.add(row)
        db.flush()
        record_audit(
            db, event_type="MANAGEMENT_CLIENT_REGISTERED", request=request, user=auth.user,
            module="management", target_type="management_client_intake", target_id=str(row.id),
            details={"client_code": row.client_code, "source_person_name": row.source_person_name},
        )
        db.commit()
    except IntegrityError as exc:
        db.rollback()
        raise HTTPException(status_code=409, detail="Client ID already exists") from exc
    except Exception:
        db.rollback()
        raise
    db.refresh(row)
    return row


@router.put("/{client_id}", response_model=ManagementClientResponse)
def update_management_client(
    client_id: int, payload: ManagementClientInput, request: Request,
    db: Session = Depends(get_db), auth: CurrentAuth = Depends(get_current_auth),
):
    _management_only(auth)
    row = db.get(ManagementClientIntake, client_id)
    if row is None:
        raise HTTPException(status_code=404, detail="Management client not found")
    _check_code(db, payload.client_code, exclude_id=row.id)
    linked_finance_client = db.scalar(select(FinanceClient).where(FinanceClient.client_code == row.client_code))
    if linked_finance_client is not None and linked_finance_client.source_team == "management":
        if payload.client_code != row.client_code:
            raise HTTPException(status_code=409, detail="Client ID cannot change after Finance has created a management project")
        # Keep the linked Finance invoice/client snapshot fields in sync without changing
        # the original management source, BD ownership or historical invoice records.
        linked_finance_client.client_name = payload.client_name
        linked_finance_client.client_email = payload.client_email
        linked_finance_client.contact_person_name = payload.contact_person_name
        linked_finance_client.contact_person_phone = payload.contact_person_phone or "Not provided"
        linked_finance_client.address = payload.address
        linked_finance_client.country = payload.country
        linked_finance_client.gst_number = payload.gst_number
        linked_finance_client.description = payload.work_description
        linked_finance_client.updated_by_id = auth.user.id
        if linked_finance_client.master_profile is not None:
            linked_finance_client.master_profile.organization_email = payload.organization_email
            linked_finance_client.master_profile.contact_person_email = payload.contact_person_email
            linked_finance_client.master_profile.task = payload.work_description
            linked_finance_client.master_profile.updated_by_id = auth.user.id
    for field, value in _values(payload).items():
        setattr(row, field, value)
    row.updated_by_id = auth.user.id
    try:
        record_audit(
            db, event_type="MANAGEMENT_CLIENT_UPDATED", request=request, user=auth.user,
            module="management", target_type="management_client_intake", target_id=str(row.id),
            details={"client_code": row.client_code, "source_person_name": row.source_person_name},
        )
        db.commit()
    except IntegrityError as exc:
        db.rollback()
        raise HTTPException(status_code=409, detail="Client ID already exists") from exc
    except Exception:
        db.rollback()
        raise
    db.refresh(row)
    return row


def _approval_flags(
    db: Session, *, link: ManagementProjectLink, workflow_status: str,
    summary: dict, viewer_user_id: int,
) -> dict:
    intake = db.get(ManagementClientIntake, link.management_client_id)
    workflow = db.get(ProjectWorkflow, link.project_id)
    approved = management_source_approval_event(db, link.project_id) is not None
    fully_delivered = (
        summary["work_units"] > 0 and summary["rework_count"] == 0
        and all(summary["stages"][stage]["completed"] == summary["work_units"]
                for stage in ("production", "qc", "qa", "delivery"))
    )
    return {
        "source_approved": approved,
        "can_approve": bool(
            intake is not None and intake.created_by_id == viewer_user_id
            and workflow is not None and workflow.operational_completed_at is not None
            and workflow_status == APPROVAL_PENDING and fully_delivered and not approved
        ),
    }


@router.post("/projects/{project_id}/approve")
def approve_management_project(
    project_id: int, request: Request,
    db: Session = Depends(get_db), auth: CurrentAuth = Depends(get_current_auth),
):
    """Only the original source may approve a fully delivered Management project."""
    _management_only(auth)
    link = db.get(ManagementProjectLink, project_id)
    if link is None:
        raise HTTPException(status_code=404, detail="Management project not found")
    intake = db.get(ManagementClientIntake, link.management_client_id)
    if intake is None or intake.created_by_id != auth.user.id:
        raise HTTPException(status_code=403, detail="Only the original Management source may approve this project")
    try:
        workflow = db.scalar(select(ProjectWorkflow).where(
            ProjectWorkflow.project_id == project_id,
        ).with_for_update())
        if workflow is None:
            raise HTTPException(status_code=404, detail="Project workflow not found")
        if workflow.status not in {APPROVAL_PENDING, FEEDBACK_NOT_SENT, READY_FOR_BILLING}:
            raise HTTPException(status_code=409, detail="Project is not awaiting Management approval")
        if workflow.operational_completed_at is None:
            raise HTTPException(status_code=409, detail="Project has not completed operational delivery")
        if management_source_approval_event(db, project_id) is not None:
            raise HTTPException(status_code=409, detail="This project is already approved")
        if db.scalar(select(ProjectInvoice.id).where(ProjectInvoice.project_id == project_id).limit(1)) is not None:
            raise HTTPException(status_code=409, detail="Existing invoices require manual review before approval")
        packages = _project_packages(db, [project_id]).get(project_id, [])
        aggregates = file_aggregates_for_packages(db, [package.id for package in packages])
        summary = _tracking_rollup(packages, aggregates)
        if not _approval_flags(db, link=link, workflow_status=APPROVAL_PENDING,
                               summary=summary, viewer_user_id=auth.user.id)["can_approve"]:
            raise HTTPException(status_code=409, detail="Production, QC, QA and Delivery must all be completed")
        _transition(
            db, workflow=workflow, actor=auth.user, status=READY_FOR_BILLING,
            event_type=MANAGEMENT_SOURCE_APPROVED_EVENT,
            title="Original Management source approved project for billing",
            comments=f"Approved by {auth.user.full_name} after Production, QC, QA and Delivery.",
        )
        record_audit(
            db, event_type="MANAGEMENT_PROJECT_SOURCE_APPROVED", request=request, user=auth.user,
            module="management", target_type="finance_project", target_id=str(project_id),
            details={"project_code": db.get(FinanceProject, project_id).project_code,
                     "management_client_id": link.management_client_id,
                     "approved_by_user_id": auth.user.id},
        )
        db.commit()
        return {"project_id": project_id, "workflow_status": READY_FOR_BILLING,
                "approved_by_user_id": auth.user.id}
    except HTTPException:
        db.rollback()
        raise
    except Exception:
        db.rollback()
        raise

@router.get("/project-tracking")
def management_client_project_tracking(
    db: Session = Depends(get_db), auth: CurrentAuth = Depends(get_current_auth),
):
    _management_only(auth)
    links = db.scalars(select(ManagementProjectLink).order_by(
        ManagementProjectLink.created_at.desc(), ManagementProjectLink.project_id.desc(),
    )).all()
    project_ids = [link.project_id for link in links]
    packages_by_project = _project_packages(db, project_ids)
    all_packages = [package for packages in packages_by_project.values() for package in packages]
    aggregates = file_aggregates_for_packages(db, [package.id for package in all_packages])
    last_activities = {}
    if project_ids:
        last_activities = dict(db.execute(select(
            WorkPackageItemActivity.project_id, func.max(WorkPackageItemActivity.updated_at),
        ).where(WorkPackageItemActivity.project_id.in_(project_ids)).group_by(
            WorkPackageItemActivity.project_id,
        )).all())

    projects = []
    by_client = {}
    for link in links:
        project = _project_payload(db, link)
        summary = _tracking_rollup(
            packages_by_project[link.project_id], aggregates,
            last_activities.get(link.project_id),
        )
        projects.append({**project, "tracking": summary, **_approval_flags(
            db, link=link, workflow_status=project["workflow_status"],
            summary=summary, viewer_user_id=auth.user.id,
        )})
        by_client.setdefault(link.management_client_id, []).append(summary)

    clients = {}
    for client_id, summaries in by_client.items():
        work_units = sum(item["work_units"] for item in summaries)
        stage_completed = sum(
            item["stages"][stage]["completed"]
            for item in summaries for stage in ("production", "qc", "qa", "delivery")
        )
        clients[str(client_id)] = {
            "project_count": len(summaries),
            "active_projects": sum(
                item["work_units"] == 0 or
                item["stages"]["delivery"]["completed"] < item["work_units"]
                for item in summaries
            ),
            "completed_projects": sum(
                item["work_units"] > 0 and
                item["stages"]["delivery"]["completed"] == item["work_units"]
                for item in summaries
            ),
            "overall_percent": round(100 * stage_completed / (4 * work_units), 1)
            if work_units else 0.0,
            "work_units": work_units,
        }
    return {"as_of": utc_now().isoformat(), "projects": projects, "clients": clients}


@router.get("/project-tracking/{project_id}")
def management_client_project_tracking_detail(
    project_id: int, db: Session = Depends(get_db),
    auth: CurrentAuth = Depends(get_current_auth),
):
    _management_only(auth)
    link = db.get(ManagementProjectLink, project_id)
    if link is None:
        raise HTTPException(status_code=404, detail="Management project not found")
    packages = _project_packages(db, [project_id]).get(project_id, [])
    package_ids = [package.id for package in packages]
    aggregates = file_aggregates_for_packages(db, package_ids)
    rows = build_item_rows(db, package_ids)
    active_packages = {row["work_package_id"] for row in rows if row["is_active"]}
    package_by_id = {package.id: package for package in packages}
    file_rows = []
    for row in rows:
        package = package_by_id[row["work_package_id"]]
        file_rows.append({
            "work_package_item_id": row["work_package_item_id"],
            "package_code": package.package_code,
            "item_number": row["item_number"], "item_name": row.get("item_name"),
            "is_active": row["is_active"], "tracking_basis": "file_activity",
            "production": "rework" if row.get("production_rework_pending") else row["production_status"],
            "qc": row["qc_status"], "qa": row["qa_status"],
            "delivery": row["delivery_status"],
            "production_completed_at": row.get("production_completed_at"),
            "qc_completed_at": row.get("qc_end"),
            "qa_completed_at": row.get("qa_end"),
            "delivered_at": row.get("delivery_at"),
            "production_employee": row.get("production_employee"),
            "qc_employee": row.get("qc_employee"),
            "qa_employee": row.get("qa_employee"),
        })
    for package in packages:
        if package.id in active_packages:
            continue
        # Area-only legacy work is counted once; inactive-only files are not.
        if any(row["work_package_id"] == package.id for row in rows):
            continue
        stage = (package.current_stage or "").lower()
        legacy = _legacy_stage_counts(package)
        file_rows.append({
            "work_package_item_id": None,
            "package_code": package.package_code,
            "item_number": package.package_code,
            "item_name": package.package_name,
            "is_active": True, "tracking_basis": "legacy_area",
            "production": "rework" if stage == "production_rework" else
                ("completed" if legacy["production"] else package.production_state),
            "qc": "passed" if legacy["qc"] else package.qc_state,
            "qa": "passed" if legacy["qa"] else package.qa_state,
            "delivery": "delivered" if legacy["delivery"] else "pending",
            "production_completed_at": package.production_completed_at.isoformat()
                if package.production_completed_at else None,
            "qc_completed_at": package.qc_submitted_at.isoformat()
                if package.qc_submitted_at else None,
            "qa_completed_at": package.qa_submitted_at.isoformat()
                if package.qa_submitted_at else None,
            "delivered_at": package.delivered_at.isoformat() if package.delivered_at else None,
            "production_employee": None, "qc_employee": None, "qa_employee": None,
        })
    activity_at = db.scalar(select(func.max(WorkPackageItemActivity.updated_at)).where(
        WorkPackageItemActivity.project_id == project_id,
    ))
    project = _project_payload(db, link)
    summary = _tracking_rollup(packages, aggregates, activity_at)
    return {
        "project_id": project_id,
        "as_of": utc_now().isoformat(),
        "project": {**project, **_approval_flags(
            db, link=link, workflow_status=project["workflow_status"],
            summary=summary, viewer_user_id=auth.user.id,
        )},
        "summary": summary,
        "files": file_rows,
    }

# The existing lifecycle dashboard's progress_percent is DELIVERY progress, not
# Production/QC/QA completion. This Finance-only read model uses the same
# per-file aggregation as the Production Tracking screen, without changing the
# operational engine, project statuses, invoices or permissions.
_TRACKING_STAGES = ("production", "qc", "qa", "delivery")


def _legacy_stage_counts(package: OrthoWorkPackage) -> dict[str, int]:
    """One legacy area = one work unit only when it has no active file children."""
    stage = (package.current_stage or "").lower()
    rework = stage == "production_rework"
    delivered = stage == "delivered" or package.delivered_at is not None
    production = (package.production_state or "").lower() == "completed" or stage in {
        "qc", "qa", "delivery_ready", "delivered",
    }
    qc = (package.qc_state or "").lower() in {"approved", "passed"} or stage in {
        "qa", "delivery_ready", "delivered",
    }
    qa = (package.qa_state or "").lower() in {"approved", "passed"} or stage in {
        "delivery_ready", "delivered",
    }
    return {
        "production": int(bool(production and not rework)),
        "qc": int(bool(qc and not rework)),
        "qa": int(bool(qa and not rework)),
        "delivery": int(bool(delivered)),
    }


def _tracking_rollup(packages: list[OrthoWorkPackage], aggregates: dict[int, dict],
                     last_activity_at=None) -> dict:
    """Stage percentages are per physical file (or legacy area), not km/quantity.

    Rework flags and file-completed counts come from the authoritative shared
    file_aggregates_for_packages function, which ignores inactive files and
    handles repeated attempts without counting the same file twice.
    """
    total = 0
    counts = dict.fromkeys(_TRACKING_STAGES, 0)
    in_progress = dict.fromkeys(_TRACKING_STAGES, 0)
    rework = 0
    active_files = 0
    legacy_areas = 0
    updated = last_activity_at
    for package in packages:
        if package.updated_at is not None and (updated is None or package.updated_at > updated):
            updated = package.updated_at
        agg = aggregates.get(package.id, {})
        expected = int(agg.get("files_expected") or 0)
        if expected:
            active_files += expected
            total += expected
            for stage, field in (("production", "files_production_done"),
                                 ("qc", "files_qc_passed"),
                                 ("qa", "files_qa_passed"),
                                 ("delivery", "files_delivered")):
                counts[stage] += int(agg.get(field) or 0)
            for stage in ("production", "qc", "qa"):
                in_progress[stage] += int(agg.get(f"{stage}_in_progress") or 0)
            # The shared aggregate can flag one rejected file in both
            # production_rework and qc_rework; count that file only once.
            rework += max(int(agg.get("production_rework") or 0),
                          int(agg.get("qc_rework") or 0)) + int(agg.get("qa_rework") or 0)
        elif int(agg.get("files_total") or 0) > 0:
            # Inactive-only file packages must not become fake legacy work units.
            continue
        else:
            legacy_areas += 1
            total += 1
            legacy = _legacy_stage_counts(package)
            for stage in _TRACKING_STAGES:
                counts[stage] += legacy[stage]
            if (package.current_stage or "").lower() == "production_rework":
                rework += 1
            if (package.current_stage or "").lower() == "production":
                in_progress["production"] += 1
    stages = {
        stage: {
            "completed": counts[stage], "total": total,
            "percent": round(100 * counts[stage] / total, 1) if total else 0.0,
            "in_progress": in_progress[stage],
        }
        for stage in _TRACKING_STAGES
    }
    overall = round(100 * sum(counts.values()) / (4 * total), 1) if total else 0.0
    if not total:
        operational_status = "Awaiting work allocation"
    elif rework:
        operational_status = "Rework in progress"
    elif counts["delivery"] == total:
        operational_status = "Delivered"
    elif counts["qa"] == total:
        operational_status = "Ready for delivery"
    elif counts["qa"] or in_progress["qa"]:
        operational_status = "QA in progress"
    elif counts["qc"] == total:
        operational_status = "Ready for QA"
    elif counts["qc"] or in_progress["qc"]:
        operational_status = "QC in progress"
    elif counts["production"] == total:
        operational_status = "Production complete - QC pending"
    elif counts["production"] or in_progress["production"]:
        operational_status = "Production in progress"
    else:
        operational_status = "Assigned - work not started"
    return {
        "overall_percent": overall, "operational_status": operational_status,
        "work_units": total, "active_files": active_files, "legacy_areas": legacy_areas,
        "rework_count": rework, "stages": stages,
        "last_activity_at": updated.isoformat() if updated is not None else None,
    }


def _project_packages(db: Session, project_ids: list[int]) -> dict[int, list[OrthoWorkPackage]]:
    if not project_ids:
        return {}
    rows = db.scalars(select(OrthoWorkPackage).where(
        OrthoWorkPackage.project_id.in_(project_ids)
    ).order_by(OrthoWorkPackage.project_id, OrthoWorkPackage.id)).all()
    result: dict[int, list[OrthoWorkPackage]] = {project_id: [] for project_id in project_ids}
    for row in rows:
        result[row.project_id].append(row)
    return result

def _amount_inr(amount: Decimal, currency: str | None, stored_inr: Decimal | None, rate: Decimal | None) -> Decimal | None:
    """Use the recorded accounting/realization FX, never a guessed live rate."""
    if stored_inr is not None:
        return Decimal(stored_inr)
    if (currency or "INR").upper() == "INR":
        return Decimal(amount)
    if rate is not None and Decimal(rate) > 0:
        return Decimal(amount) * Decimal(rate)
    return None


@router.get("/summary")
def management_project_summary(
    db: Session = Depends(get_db), auth: CurrentAuth = Depends(get_current_auth),
):
    """Management-only invoice/payment aggregates in INR, sourced from existing ledgers.

    Draft/cancelled invoices are not treated as issued revenue. Unconverted foreign
    currency rows are reported, not silently converted using an invented rate.
    """
    _finance_only(auth)
    project_ids = list(db.scalars(select(ManagementProjectLink.project_id)).all())
    zero = {"invoice_count": 0, "invoiced_inr": 0.0, "received_inr": 0.0,
            "outstanding_inr": 0.0, "missing_fx_rows": 0}
    if not project_ids:
        return {"projects": {}, "totals": {**zero, "collection_rate": 0.0}}

    invoices = list(db.scalars(select(ProjectInvoice).where(ProjectInvoice.project_id.in_(project_ids))).all())
    invoice_ids = [row.id for row in invoices]
    payments = list(db.scalars(select(ProjectInvoicePayment).where(
        ProjectInvoicePayment.invoice_id.in_(invoice_ids)
    )).all()) if invoice_ids else []
    by_invoice: dict[int, list[ProjectInvoicePayment]] = {}
    for payment in payments:
        by_invoice.setdefault(payment.invoice_id, []).append(payment)
    stats: dict[int, dict] = {pid: {**zero, "settled_inr": 0.0} for pid in project_ids}
    for invoice in invoices:
        target = stats[invoice.project_id]
        target["invoice_count"] += 1
        if invoice.status in {"INVOICE_DRAFT", "CANCELLED", "CANCELED"}:
            continue
        currency = (invoice.currency or "INR").upper()
        total = Decimal(invoice.amount or 0) + Decimal(invoice.tax_amount or 0)
        invoice_total_inr = _amount_inr(total, currency, invoice.total_inr, invoice.fx_rate_to_inr)
        if invoice_total_inr is None:
            target["missing_fx_rows"] += 1
        else:
            target["invoiced_inr"] += float(invoice_total_inr)
        paid_at_invoice_rate = Decimal("0")
        paid_is_complete = True
        for payment in by_invoice.get(invoice.id, []):
            paid_inr = _amount_inr(
                Decimal(payment.amount), payment.payment_currency or currency,
                payment.inr_equivalent, payment.fx_rate_to_inr,
            )
            if paid_inr is None:
                target["missing_fx_rows"] += 1
            else:
                target["received_inr"] += float(paid_inr)
            applied_inr = _amount_inr(
                Decimal(payment.amount), currency,
                payment.invoice_inr_equivalent, invoice.fx_rate_to_inr,
            )
            if applied_inr is None:
                paid_is_complete = False
            else:
                paid_at_invoice_rate += applied_inr
        if invoice_total_inr is not None and paid_is_complete:
            applied = min(invoice_total_inr, paid_at_invoice_rate)
            target["settled_inr"] += float(applied)
            target["outstanding_inr"] += float(max(Decimal("0"), invoice_total_inr - applied))
        elif invoice_total_inr is not None:
            target["missing_fx_rows"] += 1

    totals = {key: sum(item[key] for item in stats.values()) for key in
              ("invoice_count", "invoiced_inr", "received_inr", "outstanding_inr", "missing_fx_rows", "settled_inr")}
    totals["collection_rate"] = round(
        100 * totals["settled_inr"] / totals["invoiced_inr"], 1
    ) if totals["invoiced_inr"] else 0.0
    for item in stats.values():
        item.pop("settled_inr")
    totals.pop("settled_inr")
    return {"projects": {str(key): value for key, value in stats.items()}, "totals": totals}

function managementProgressTone(p: number): 'red'|'orange'|'yellow'|'green' {
  const n = Number.isFinite(p) ? Math.max(0, Math.min(100, p)) : 0
  if (n < 30) return 'red'
  if (n < 50) return 'orange'
  if (n < 70) return 'yellow'
  return 'green'
}

const statusTone = (value: string) => {
  const s = value.toLowerCase()
  if (['completed', 'passed', 'approved', 'delivered'].includes(s)) return 'complete'
  if (['rejected', 'rework', 'production_rework'].includes(s)) return 'rework'
  if (s === 'in_progress' || s === 'started') return 'active'
  return 'pending'
}

// Management UI presentation only. Never change incoming percentages.
function managementProgressTone(p: number): 'red'|'orange'|'yellow'|'green' {
  const n = Number.isFinite(p) ? Math.max(0, Math.min(100, p)) : 0
  if (n < 30) return 'red'
  if (n < 50) return 'orange'
  if (n < 70) return 'yellow'
  return 'green'
}

      return <div className="mp-stage-track" key={stage}>
        <div className="mp-stage-track-label"><strong>{STAGE_LABELS[stage]}</strong><span>{item.completed}/{item.total} · {item.percent.toFixed(0)}%</span></div>
        <div className="mp-progress"><span className={`mp-v8-${managementProgressTone(item.percent)}`} style={{ width: `${Math.min(100, Math.max(0, item.percent))}%` }}/></div>
      </div>
    })}
  </div>
}

const currentWorkflowStep = (status: string) => {
  const s = status.toUpperCase()
  if (s === 'CLOSED') return 7
  if (/PAYMENT|INVOICE_RAISED|OVERDUE/.test(s)) return 6
  if (/INVOICE_DRAFT|READY_FOR_BILLING/.test(s)) return 5
  if (/DELIVER|APPROVAL_PENDING|CLIENT_FEEDBACK|CLIENT_ACCEPT/.test(s)) return 4
  if (/\bQA\b|\bQC\b|QUALITY/.test(s)) return 3
  if (/PRODUCTION|IN_PROGRESS|WORK/.test(s)) return 2
  if (/PM_ASSIGNED|ASSIGNED/.test(s)) return 1
  return 0
}
function exportCsv(rows: ManagementProject[], clients: ManagementClient[], lifecycle: LifecycleDashboard | null, summary: Summary | null, tracking: TrackingResponse | null) {
  const escapeCell = (value: string | number | null | undefined) => {
    const raw = String(value ?? '')
    // Avoid formula execution when the CSV is opened in Excel.
    const safe = /^[\s]*[=+@\-\t\r]/.test(raw) && typeof value === 'string' ? `'${raw}` : raw
    return `"${safe.replace(/"/g, '""')}"`
  }
  const columns = ['Project Code', 'Project Name', 'Client', 'Source', 'Department', 'Project Manager', 'Status', 'Operational Status', 'Overall Stage Progress %', 'Production %', 'QC %', 'QA %', 'Delivery %', 'Project Value', 'Currency', 'Invoiced INR', 'Received INR', 'Outstanding INR']
  const csv = [columns.map(escapeCell).join(','), ...rows.map(row => {
    const client = clients.find(item => item.id === row.management_client_id)
    const life = lifecycle?.projects.find(item => item.id === row.id)
    const financial = summary?.projects[String(row.id)] ?? EMPTY_FINANCIAL
    const stages = tracking?.projects[String(row.id)]
    return [row.project_code, row.project_name, client?.client_name, row.source_person_name, row.department_label,
      row.project_manager_name, life?.workflow_status ?? row.workflow_status,
      stages?.operational_status ?? 'Tracking unavailable', stages?.overall_percent ?? '',
      ...STAGE_KEYS.map(stage => stages?.stages[stage].percent ?? ''),
      row.project_value, row.currency, financial.invoiced_inr, financial.received_inr, financial.outstanding_inr].map(escapeCell).join(',')
  })].join('\r\n')
  const url = URL.createObjectURL(new Blob(['\uFEFF', csv], { type: 'text/csv;charset=utf-8;' }))
  const link = document.createElement('a')
  link.href = url
  link.download = 'nakshatech-management-projects.csv'
  document.body.appendChild(link)
  link.click()
  link.remove()
  window.setTimeout(() => URL.revokeObjectURL(url), 1000)
}
